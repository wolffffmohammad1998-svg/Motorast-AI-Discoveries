import sys; sys.setrecursionlimit(10000)
def has_pow2_cycle(n, edges):
    adj=[[] for _ in range(n)]
    for u,v in edges: adj[u].append(v); adj[v].append(u)
    assert min(len(a) for a in adj) >= 3, "min degree < 3"
    lens={2**k for k in range(2,12) if 2**k<=n}
    for s in range(n):
        vis=[False]*n; vis[s]=True
        def dfs(v,d):
            for w in adj[v]:
                if w==s and d in lens and d>=3: return True
                if w>s and not vis[w]:
                    vis[w]=True
                    if dfs(w,d+1): return True
                    vis[w]=False
            return False
        if dfs(s,1): return True
    return False
# اگر False بود و درجه ≥۳: مثال نقض است. پیش از هر ادعا، ریاضی‌دان‌ها بررسی کنند.
